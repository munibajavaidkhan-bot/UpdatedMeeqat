<script>
// Counter Animation
document.addEventListener('DOMContentLoaded', function() {
    const target = {{ $result['fabric_meters'] }};
    const el     = document.getElementById('metersCounter');

    let current  = 0;
    const step   = target / 50;
    const timer  = setInterval(() => {
        current += step;
        if (current >= target) {
            el.textContent = target.toFixed(2);
            clearInterval(timer);
        } else {
            el.textContent = current.toFixed(2);
        }
    }, 30);
});

// Copy Result
function copyResult() {
    const text = `Meeqat.io — Chaddar Size Calculator\n\n` +
                 `Height: {{ $result['height_cm'] }} cm ({{ $result['height_feet'] }})\n` +
                 `Style: {{ $result['style_label'] }}\n` +
                 `Size: {{ $result['size_label'] }}\n` +
                 `Fabric Required: {{ $result['fabric_meters'] }} Meters\n\n` +
                 `Calculate your size: ${window.location.origin}{{ route('calculator.chaddar') }}`;

    navigator.clipboard.writeText(text).then(() => {
        const btn = document.getElementById('copyBtn');
        const original = btn.innerHTML;
        btn.innerHTML = `<svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M5 13l4 4L19 7"/></svg> Copied!`;
        btn.classList.add('text-primary-400', 'border-primary-500/40');
        setTimeout(() => {
            btn.innerHTML = original;
            btn.classList.remove('text-primary-400', 'border-primary-500/40');
        }, 3000);
    });
}

// Share Result
function shareResult() {
    const shareData = {
        title: 'Meeqat.io — Chaddar Size Calculator',
        text: `Meri height {{ $result['height_cm'] }} cm hai. Mujhe {{ $result['fabric_meters'] }} meters fabric chahiye (Size {{ $result['size_label'] }}). Apna size check karein:`,
        url: '{{ route("calculator.chaddar") }}'
    };

    if (navigator.share) {
        navigator.share(shareData);
    } else {
        copyResult();
        window.dispatchEvent(new CustomEvent('toast', {
                detail: { message: 'Link copied! Now share it.', type: 'success' }
        }));
    }
}
</script>
